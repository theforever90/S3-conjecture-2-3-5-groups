# Code 3: Rational Groups Between $A_5 \times A_5$ and $\mathrm{Aut}(A_5 \times A_5)$

## Purpose

Let $R = A_5 \times A_5$. This code enumerates **all groups** $G$ satisfying $R \leq G \leq \mathrm{Aut}(R)$ and filters those that are **rational groups**. It also saves the permutation-group constructions of all rational groups to a GAP-readable file. For each rational group found, we compute:

1. **Conjugacy class sizes** — both the full sorted list and the set of distinct sizes
2. **Centralizer sizes** $|C_G(g)|$ with corresponding element orders
3. **Structure description**
4. **Permutation-group construction** — saved as `code3_rational_groups.g`

---

## Code

```gap
# ==============================================================================
# Utility Functions
# ==============================================================================

# A group G is rational if every element is conjugate to all its
# powers that are coprime to its order.
IsRationalGroup := function(G)
    local classes, rep, order, k;
    classes := ConjugacyClasses(G);
    for rep in List(classes, Representative) do
        order := Order(rep);
        for k in [1..order] do
            if Gcd(k, order) = 1 then
                if not IsConjugate(G, rep, rep^k) then
                    return false;
                fi;
            fi;
        od;
    od;
    return true;
end;

# Conjugacy class sizes (full list, sorted)
CCS := function(G)
    local sizes;
    sizes := List(ConjugacyClasses(G), Size);
    Sort(sizes);
    return sizes;
end;

# Conjugate type vector (distinct sizes only)
CTV := function(G)
    return Set(List(ConjugacyClasses(G), Size));
end;

# Centralizer sizes with element orders, sorted by |C|
SCOE := function(G)
    local data;
    data := List(ConjugacyClasses(G),
        c -> [Order(Representative(c)), Int(Size(G)/Size(c))]);
    Sort(data, function(a,b) return a[2] < b[2]; end);
    return data;
end;

# Save the rational groups as permutation groups in a GAP-readable file.
# After running this script, use:
#     Read("code3_rational_groups.g");
# to reload Code3RationalGroup1, ..., and Code3RationalGroups.
SavePermutationGroupRecords := function(records, filename)
    local i, G, gens, j, varname;

    PrintTo(filename,
        "# Permutation-group constructions for Code 3\n",
        "# Generated by the GAP code in Code 3.md\n\n");

    for i in [1..Length(records)] do
        G := records[i].group;
        gens := GeneratorsOfGroup(G);
        varname := Concatenation("Code3RationalGroup", String(i));

        AppendTo(filename, "# Rational group #", i, "\n");
        AppendTo(filename, "# order = ", Size(G), "\n");
        AppendTo(filename, "# candidate index = ", records[i].candidate_index, "\n");
        AppendTo(filename, "# index in Aut(A5xA5) = ", records[i].index_in_aut, "\n");
        AppendTo(filename, "# structure = ", StructureDescription(G), "\n");
        AppendTo(filename, varname, " := Group(\n");

        for j in [1..Length(gens)] do
            AppendTo(filename, "    ", gens[j]);
            if j < Length(gens) then
                AppendTo(filename, ",\n");
            else
                AppendTo(filename, "\n");
            fi;
        od;

        AppendTo(filename, ");\n");
        AppendTo(filename, "SetName(", varname, ", \"", varname, "\");\n\n");
    od;

    AppendTo(filename, "Code3RationalGroups := [\n");
    for i in [1..Length(records)] do
        AppendTo(filename, "    Code3RationalGroup", i);
        if i < Length(records) then
            AppendTo(filename, ",\n");
        fi;
    od;
    AppendTo(filename, "\n];\n\n");

    AppendTo(filename, "Code3RationalGroupRecords := [\n");
    for i in [1..Length(records)] do
        AppendTo(filename, "    rec(\n");
        AppendTo(filename, "        group := Code3RationalGroup", i, ",\n");
        AppendTo(filename, "        order := ", records[i].order, ",\n");
        AppendTo(filename, "        candidate_index := ", records[i].candidate_index, ",\n");
        AppendTo(filename, "        index_in_aut := ", records[i].index_in_aut, ",\n");
        AppendTo(filename, "        structure := \"",
                 StructureDescription(records[i].group), "\"\n");
        AppendTo(filename, "    )");
        if i < Length(records) then
            AppendTo(filename, ",\n");
        else
            AppendTo(filename, "\n");
        fi;
    od;
    AppendTo(filename, "];\n\n");

    AppendTo(filename,
        "Print(\"Loaded rational permutation groups from Code 3.\\n\");\n");
end;

# ==============================================================================
# Construct M = A5 x A5 and A = Aut(A5 x A5)
# ==============================================================================

# Permutation representation on 10 points:
#   {1..5}  -> first A5 (or S5)
#   {6..10} -> second A5 (or S5)

a5_1 := Group((1,2,3,4,5), (1,2,3));
a5_2 := Group((6,7,8,9,10), (6,7,8));
M := ClosureGroup(a5_1, a5_2);              # M = A5 x A5, |M| = 3600

s5_1 := Group((1,2,3,4,5), (1,2));
s5_2 := Group((6,7,8,9,10), (6,7));

# Coordinate-swapping involution
tau := (1,6)(2,7)(3,8)(4,9)(5,10);

# A = (S5 x S5) ⋊ <tau> ≅ Aut(A5 x A5), |A| = 28800
A := ClosureGroup(ClosureGroup(s5_1, s5_2), Group(tau));

Print("M = A5 x A5,  |M| = ", Size(M), "\n");
Print("A = Aut(A5xA5), |A| = ", Size(A), "\n");
Print("Index [A:M] = ", Size(A)/Size(M), "\n\n");

# ==============================================================================
# Find All Intermediate Subgroups and Filter Rational Groups
# ==============================================================================

Print("Computing intermediate subgroups...\n");
inter := IntermediateSubgroups(A, M);
all_candidates := Concatenation([M], inter.subgroups, [A]);
Print("Total candidate groups (including M and A): ",
      Length(all_candidates), "\n\n");

Print("Filtering rational groups...\n");
rational_groups := [];
candidate_index := 0;
for G in all_candidates do
    candidate_index := candidate_index + 1;
    if IsRationalGroup(G) then
        Add(rational_groups, rec(
            group := G,
            order := Size(G),
            candidate_index := candidate_index,
            index_in_aut := Index(A, G),
            generators := GeneratorsOfGroup(G)
        ));
        Print("  Found rational group of order ", Size(G),
              " (candidate index ", candidate_index, ")\n");
    fi;
od;
Sort(rational_groups, function(a,b) return a.order < b.order; end);
Print("\nTotal rational groups found: ", Length(rational_groups), "\n");

save_filename := "code3_rational_groups.g";
SavePermutationGroupRecords(rational_groups, save_filename);
Print("Saved permutation-group constructions to ", save_filename, "\n");

# ==============================================================================
# Detailed Analysis of Each Rational Group
# ==============================================================================

for i in [1..Length(rational_groups)] do
    G := rational_groups[i].group;
    ord := Size(G);
    idx := Index(A, G);

    Print("\n========================================\n");
    Print("Rational Group #", i, "\n");
    Print("========================================\n");
    Print("Order:               ", ord, "\n");
    Print("Candidate index:     ", rational_groups[i].candidate_index, "\n");
    Print("Index in Aut(A5xA5): ", idx, "\n");
    Print("Structure:           ", StructureDescription(G), "\n");
    Print("Generators:          ", GeneratorsOfGroup(G), "\n");

    ccs := CCS(G);
    Print("Number of conjugacy classes: ", Length(ccs), "\n");
    Print("Conjugacy class sizes (sorted):\n  ", ccs, "\n");

    ctv := CTV(G);
    Print("Distinct sizes (", Length(ctv), " values):\n  ", ctv, "\n");

    scoe := SCOE(G);
    Print("(Element order, |C|) sorted by |C|:\n");
    for pair in scoe do
        Print("  (", String(pair[1],2), ", ", String(pair[2],6), ")");
        if pair[2] in [4, 6, 8, 9, 10, 15] then
            Print("  <-- key size");
        fi;
        Print("\n");
    od;
    Print("========================================\n");
od;
```
